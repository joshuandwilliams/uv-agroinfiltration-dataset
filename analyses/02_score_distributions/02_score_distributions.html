# 2.1
# The original scoring key is a CMYK JPEG, so an sRGB copy (converted via its
# ICC profile) is read here to keep colours faithful.
key <- readPNG(here("analyses", "02_score_distributions", "assets", "lesion_score_key_rgb.png"))
key_raster <- as.raster(key[189:554, , 1:3])
W <- dim(key)[2] # key width (px)
key_h <- length(189:554) # cropped key height (panels only)

# key panel edges (measured) -> example column boxes match these exactly
L <- c(7, 404, 801, 1195, 1586, 1978, 2369)
R <- c(389, 786, 1180, 1571, 1963, 2354, 2745)
cen <- (L + R) / 2

# Four random example CDAs per score (0-6)
set.seed(42)
n_per <- 4
thumbs <- lapply(0:6, function(sc) {
  files <- list.files(here("data", "cropped_images", sc), full.names = TRUE, pattern = "\\.tif$")
  lapply(sample(files, n_per), function(f) as.raster(readTIFF(f)))
})

# Layout in pixel units (bottom to top): key panels, "Scoring Key" title, grid of boxed examples, score labels.
# Boxes are centred on the key panels but a little narrower, leaving clear gaps between columns.
box_hw <- 160
pad <- 16
tgap <- 12
side <- 2 * box_hw - 2 * pad
title_band <- 150
lab_band <- 150
box_h <- n_per * side + (n_per - 1) * tgap + 2 * pad
g0 <- key_h + title_band # grid (box) bottom
g1 <- g0 + box_h # grid (box) top
Htot <- g1 + lab_band

boxes <- data.frame(xmin = cen - box_hw, xmax = cen + box_hw, ymin = g0, ymax = g1)

thumb_layers <- list()
for (j in 1:7) {
  for (i in seq_len(n_per)) {
    ytop <- g1 - pad - (i - 1) * (side + tgap)
    thumb_layers <- c(
      thumb_layers,
      annotation_raster(thumbs[[j]][[i]],
        xmin = cen[j] - side / 2, xmax = cen[j] + side / 2,
        ymin = ytop - side, ymax = ytop, interpolate = TRUE
      )
    )
  }
}

p <- ggplot() +
  annotation_raster(key_raster, xmin = 0, xmax = W, ymin = 0, ymax = key_h, interpolate = TRUE) +
  annotate("text", x = W / 2, y = key_h + title_band / 2, label = "Scoring Key", size = 7) +
  thumb_layers +
  geom_rect(
    data = boxes, aes(xmin = xmin, xmax = xmax, ymin = ymin, ymax = ymax),
    fill = NA, colour = "black", linewidth = 0.7
  ) +
  annotate("text", x = cen, y = g1 + lab_band / 2, label = 0:6, size = 8) +
  coord_fixed(xlim = c(0, W), ylim = c(0, Htot), expand = FALSE) +
  theme_void()

ggsave(here("analyses", "02_score_distributions", "thesis_plots", "cdaexamples.png"),
  p,
  width = W / 300, height = Htot / 300, dpi = 300, device = ragg::agg_png, bg = "white"

# Largest-remainder rounding so displayed percentages sum to exactly 100.0
pct_sum_100 <- function(counts, digits = 1) {
  scaled <- counts / sum(counts) * 100 * 10^digits
  floor_vals <- floor(scaled)
  deficit <- round(100 * 10^digits) - sum(floor_vals)
  top_up <- order(-(scaled - floor_vals))[seq_len(deficit)]
  floor_vals[top_up] <- floor_vals[top_up] + 1
  floor_vals / 10^digits
}

all_counts <- all_counts %>% mutate(Pct = pct_sum_100(Count))
median_counts <- median_counts %>% mutate(Pct = pct_sum_100(Count))

checked_combined <- bind_rows(checked_1, checked_2)

# Raters were instructed to skip cell death areas belonging to the
# non-target infiltration in dual-infiltration images (Projects C and H),
# which is the primary reason for skipped scores - so skip counts are only
# interpretable once split by whether the image was a dual-infiltration one.
project_key <- load_project_key()
dual_infiltration_projects <- project_key$Project[project_key$Dual_Infiltration == 1]

image_dual_infiltration <- read_csv(
  here("analyses", "01_generate_raw_data", "01_allocation_info", "image_info.csv"),
  show_col_types = FALSE
) %>%
  fill(Path, .direction = "down") %>%
  transmute(
    Basename = Image,
    Dual_Infiltration_Image = factor(
      sub("/.*", "", Path) %in% dual_infiltration_projects,
      levels = c(FALSE, TRUE), labels = c("No", "Yes")
    )
  )

checked_long <- long_score_table(checked_combined, include_median_score = FALSE, drop_missing_scores = FALSE) %>%
  left_join(image_dual_infiltration, by = "Basename")

skips_by_scorer <- checked_long %>%
  filter(is.na(Score)) %>%
  count(Scorer, Dual_Infiltration_Image, name = "Count", .drop = FALSE)

n_skipped_per_cda <- checked_combined %>%
  left_join(image_dual_infiltration, by = "Basename") %>%
  mutate(N_Skipped = factor(is.na(Score1) + is.na(Score2) + is.na(Score3), levels = 0:3)) %>%
  count(N_Skipped, Dual_Infiltration_Image, name = "Count", .drop = FALSE)

dual_infiltration_pal <- c("No" = "steelblue", "Yes" = "darkorange")

p_scorer <- ggplot(skips_by_scorer, aes(x = Scorer, y = Count, fill = Dual_Infiltration_Image)) +

5. Interpretation and Discussion

Cell death area source images from past Banfield lab projects were annotated with bounding boxes and severity scores, which ranged from 0 (no cell death) to 6 (strong cell death). Each cell death area was annotated by three raters, and the median of these three raters’ scores are chosen as representative labels.
