Pairwise Agreement and Per-Image Analysis

Date: 15/5/2024 – 10/6/2024

GOHREP

Goal:

Investigate changes to average pairwise agreement when each scorer is removed. Investigate differences in agreement across raw agroinfiltration images.

Hypothesis:

Excluding scores from scorers with low pairwise agreement will increase the overall average pairwise agreement. Some images will be scored more consistently than others due to differences in clarity or familiarity.

Rationale:

Differences in lighting or leaf clarity could cause some images to be scored less consistently. Identifying which scorers most reduce overall agreement informs future data collection decisions.

Experimental plan:

Compute pairwise % agreement between all scorer pairs, visualise as a heatmap (with and without median-0 CDAs), count shared CDAs per pair, compute leave-one-out average agreement for each scorer, verify with random sampling, and compute the Jaccard index and agreement pattern plots.

long_score_table <- function(data) {
  long <- data %>%
    select(Basename, Row, Col, Pos, Median_Score,
           Scorer1, Score1, Scorer2, Score2, Scorer3, Score3) %>%
    pivot_longer(
      cols          = c(Scorer1, Score1, Scorer2, Score2, Scorer3, Score3),
      names_to      = c(".value", "num"),
      names_pattern = "(Scorer|Score)(\\d+)"
    ) %>%
    select(-num) %>%
    filter(!is.na(Scorer), Score != "[]")
  long$Scorer <- factor(long$Scorer, levels = sort(as.numeric(unique(long$Scorer))))
  return(long)
}

pairwise_score_data <- function(score_data_long) {
  data <- score_data_long %>%
    mutate(Score = as.numeric(Score)) %>%
    group_by(Basename, Row, Col, Pos) %>%
    mutate(id = row_number()) %>%
    ungroup()

  pairwise_comparisons <- data %>%
    full_join(data, by = c("Basename", "Row", "Col", "Pos"), relationship = "many-to-many") %>%
    filter(id.x < id.y) %>%
    select(Scorer.x, Score.x, Scorer.y, Score.y) %>%
    group_by(Scorer.x, Score.x, Scorer.y, Score.y) %>%
    summarise(Count = n(), .groups = "drop")

  filtered_data <- pairwise_comparisons %>%
    filter(!is.na(Score.x) & !is.na(Score.y))

  filtered_copy <- filtered_data
  names(filtered_copy)[names(filtered_copy) %in% c("Scorer.x", "Scorer.y")] <- c("Scorer.y", "Scorer.x")
  filtered_data <- rbind(filtered_data, filtered_copy)

  agreement_data <- filtered_data %>%
    group_by(Scorer.x, Scorer.y) %>%
    summarise(Total        = sum(Count),
              Agree        = sum(Count * (Score.x == Score.y)),
              Percent_Agree = (Agree / Total) * 100,
              .groups = "drop") %>%
    filter(as.integer(Scorer.x) < as.integer(Scorer.y))

  all_combinations <- expand.grid(Scorer.x = unique(agreement_data$Scorer.x),
                                  Scorer.y = unique(agreement_data$Scorer.y))

  all_combinations %>%
    left_join(agreement_data, by = c("Scorer.x", "Scorer.y")) %>%
    mutate(across(c(Total, Agree, Percent_Agree), ~ ifelse(is.na(.), NA, .))) %>%
    filter(as.integer(Scorer.x) < as.integer(Scorer.y))
}

average_pairwise_agreement <- function(data) {
  valid <- data[!is.na(data$Total) & !is.na(data$Percent_Agree), ]
  sum(valid$Percent_Agree * valid$Total) / sum(valid$Total)
}

4. Effect of Removing Each Scorer on Average Pairwise Agreement

remove_scorer <- function(score_data_long, scorer, remove_2_disagree) {
  subset_score_data <- subset(score_data_long, Scorer != scorer)

  if (remove_2_disagree) {
    groups_with_issues <- subset_score_data %>%
      group_by(Basename, Row, Col, Pos) %>%
      filter(n() == 2, n_distinct(Score) > 1) %>%
      ungroup() %>%
      select(Basename, Row, Col, Pos)

    subset_score_data <- subset_score_data %>%
      anti_join(groups_with_issues, by = c("Basename", "Row", "Col", "Pos"))
  }
  return(subset_score_data)
}

unique_scorers <- c("Original", as.character(sort(as.numeric(unique(score_data_long$Scorer)))))

Why does pairwise agreement increase for most scorer removals?

When a scorer is removed, CDAs that now have only 2 remaining scores are also removed — but only if those 2 scores disagree. Since disagreement occurs ~36% of the time, each scorer removal also removes a further 36% of the CDAs that scorer uniquely shared, biasing the remaining pool toward agreement.

random_sample_pair_agree <- function(data, sample_num) {
  set.seed(42)
  CDA_set    <- distinct(data, Basename, Row, Col, Pos)
  CDA_set    <- CDA_set[sample(nrow(CDA_set)), ]
  sample_num <- min(sample_num, nrow(CDA_set))

  agreement_vector <- vapply(seq_len(sample_num), function(i) {
    sub <- data %>%
      filter(Basename == CDA_set$Basename[i], Row == CDA_set$Row[i],
             Col == CDA_set$Col[i], Pos == CDA_set$Pos[i])
    if (nrow(sub) == 3) sub <- sub[-sample(1:3, 1), ]
    as.numeric(nrow(sub) == 2 && sub$Score[1] == sub$Score[2])
  }, numeric(1))

  mean(agreement_vector)
}

6. Jaccard Index Between Scorer Pairs

Measures how often each pair of scorers scored the same CDAs relative to all CDAs either of them scored. High Jaccard = the pair scored mostly the same images; low Jaccard = they scored mostly different images.

Code 
for_jaccard <- score_data_long %>%
  mutate(ObjectID = paste(Basename, Row, Col, Pos, sep = "_"))

scorer_objects <- for_jaccard %>%
  distinct(Scorer, ObjectID) %>%
  group_by(Scorer) %>%
  summarise(Objects = list(ObjectID), .groups = "drop") %>%
  mutate(Scorer = as.numeric(as.character(Scorer)))

jaccard_index <- function(o1, o2) {
  n_inter <- length(intersect(o1, o2))
  n_union <- length(union(o1, o2))
  if (n_union == 0 || n_inter == 0) NA_real_ else n_inter / n_union
}

pairs <- expand.grid(Scorer1 = scorer_objects$Scorer, Scorer2 = scorer_objects$Scorer) %>%
  filter(Scorer1 < Scorer2)

  rowwise() %>%

    scorer_objects$Objects[scorer_objects$Scorer == Scorer1][[1]],
    scorer_objects$Objects[scorer_objects$Scorer == Scorer2][[1]]

gen_pairwise_counts <- function(score_data_long) {
  data <- score_data_long %>%
    mutate(Score = as.numeric(Score)) %>%
    group_by(Basename, Row, Col, Pos) %>%
    mutate(id = row_number()) %>%
    ungroup()

  pairwise <- data %>%
    full_join(data, by = c("Basename", "Row", "Col", "Pos"), relationship = "many-to-many") %>%
    filter(id.x < id.y) %>%
    select(Scorer.x, Score.x, Scorer.y, Score.y) %>%
    group_by(Scorer.x, Score.x, Scorer.y, Score.y) %>%
    summarise(Count = n(), .groups = "drop") %>%
    filter(!is.na(Score.x) & !is.na(Score.y))

  sym_copy <- pairwise
  names(sym_copy)[names(sym_copy) %in% c("Scorer.x", "Scorer.y")] <- c("Scorer.y", "Scorer.x")
  names(sym_copy)[names(sym_copy) %in% c("Score.x",  "Score.y")]  <- c("Score.y",  "Score.x")
  rbind(pairwise, sym_copy) %>%
    mutate(Count_Binned = cut(Count,
                              breaks = c(0, 1, 10, 100, 250, Inf),
                              labels = c("1", "2-10", "11-100", "101-250", "251-1000"),
                              include.lowest = TRUE))
}

make_patterns_plot <- function(sdl) {
  counts     <- gen_pairwise_counts(sdl)
  levels_xy  <- as.character(1:10)
  counts <- counts %>%
    mutate(Scorer.x = factor(Scorer.x, levels = levels_xy),
           Scorer.y = factor(Scorer.y, levels = levels_xy))
  complete_counts <- complete(counts,
                              Scorer.x = factor(levels_xy, levels = levels_xy),
                              Scorer.y = factor(levels_xy, levels = levels_xy),
                              fill = list(Count = 0)) %>%
    mutate(Scorer.y = fct_rev(Scorer.y))
  real_counts <- complete_counts %>% filter(Count > 0)

  ggplot(complete_counts, aes(x = Score.x, y = Score.y)) +
    geom_tile(aes(fill = Count_Binned), color = "black", linewidth = 0.5) +
    geom_abline(intercept = 0, slope = 1, linetype = "dashed", color = "grey40") +
    scale_fill_manual(values = c("1"        = "#eff3ff",
                                 "2-10"     = "#9ecae1",
                                 "11-100"   = "#3182bd",
                                 "101-250"  = "#fdae61",
                                 "251-1000" = "#d73027"),
                      name = "Frequency", na.value = "white", drop = FALSE) +
    facet_grid(Scorer.y ~ Scorer.x, switch = "both") +
    theme_bw() +
    labs(x = "Scorer X", y = "Scorer Y") +
    theme(panel.spacing    = unit(0.25, "lines"),
          strip.placement  = "outside",
          panel.grid       = element_blank(),
          strip.text       = element_text(size = 11),

          axis.title       = element_text(size = 13),
          legend.title     = element_text(size = 13),
          legend.text      = element_text(size = 11))
}

8. Per-Scorer Agreement with Median

Stacked bar chart showing each scorer’s exact agreement and within-±1 agreement with the CDA median score, computed across all CDAs scored by that individual. Adapted from AutoCDAScorer_Models/analyses/0014_Scoring_Consistency/scoring_consistency.qmd (human-only, no model results).

Code 
scorer_accuracy <- score_data_long %>%
  filter(!is.na(Score), Score != "[]") %>%
  mutate(Score = as.integer(Score)) %>%
  group_by(Scorer) %>%
  summarise(
    exact      = mean(Score == Median_Score, na.rm = TRUE),
    within_one = mean(abs(Score - Median_Score) <= 1, na.rm = TRUE),
    .groups = "drop"
  ) %>%
  mutate(extra_within_one = within_one - exact)

avg_row <- scorer_accuracy %>%
  summarise(Scorer         = factor("Avg"),
            exact          = mean(exact),
            within_one     = mean(within_one),
            extra_within_one = mean(extra_within_one))

plot_acc <- bind_rows(scorer_accuracy, avg_row) %>%
  mutate(Scorer = factor(Scorer,
                         levels = c(as.character(sort(as.numeric(
                           unique(as.character(scorer_accuracy$Scorer))))), "Avg"))) %>%
  pivot_longer(cols = c(exact, extra_within_one),
               names_to = "metric", values_to = "value") %>%
  mutate(metric = factor(metric,
                         levels = c("extra_within_one", "exact"),
                         labels = c("Additional within ±1", "Exact agreement")))

p <- ggplot(plot_acc, aes(x = Scorer, y = value, fill = metric)) +
  geom_bar(stat = "identity") +
  scale_fill_manual(values = c("Exact agreement"      = "steelblue",
                               "Additional within ±1" = "cornflowerblue"),
                    name = NULL,
                    breaks = c("Exact agreement", "Additional within ±1")) +
  scale_y_continuous(labels = scales::percent, limits = c(0, 1)) +
  labs(x = "Scorer", y = "Agreement with Median Score") +
  theme_minimal() +
  theme(axis.text    = element_text(size = 13),
        axis.title   = element_text(size = 15),
        legend.title = element_text(size = 13),
        legend.text  = element_text(size = 11))
