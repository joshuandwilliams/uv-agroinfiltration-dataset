Inter-Rater Agreement

Date: 15/5/2024 – 10/6/2024

GOHREP

Goal:

Investigate changes to average pairwise agreement when each rater is removed. Investigate differences in agreement across source agroinfiltration images. Quantify the overall inter-rater reliability of the human rater panel using Krippendorff’s alpha.

Hypothesis:

Excluding scores from raters with low pairwise agreement will increase the overall average pairwise agreement. Some images will be scored more consistently than others due to differences in clarity or familiarity. Krippendorff’s alpha on raw scores will be high (>0.8), reflecting that raters broadly agree on severity ordering even when they differ by ±1.

Rationale:

Differences in lighting or leaf clarity could cause some images to be scored less consistently. Identifying which raters most reduce overall agreement informs future data collection decisions. Krippendorff’s alpha is the standard metric for ordinal inter-rater reliability and handles missing data (not all raters rated all CDAs), giving a single overall reliability estimate that pairwise % agreement alone does not.

6. Agreement Patterns Between Rater Pairs

Code 
# 6
gen_pairwise_counts <- function(score_data_long) {
  data <- score_data_long %>%
    mutate(Score = as.numeric(Score)) %>%
    group_by(Basename, Row, Col, Pos) %>%
    mutate(id = row_number()) %>%
    ungroup()

  pairwise <- data %>%
    full_join(data, by = c("Basename", "Row", "Col", "Pos"), relationship = "many-to-many") %>%
    filter(id.x < id.y) %>%
    select(Scorer.x, Score.x, Scorer.y, Score.y) %>%
    group_by(Scorer.x, Score.x, Scorer.y, Score.y) %>%
    summarise(Count = n(), .groups = "drop") %>%
    filter(!is.na(Score.x) & !is.na(Score.y))

  # id.x < id.y reflects column position (Scorer1/2/3), not rater ID value,
  # so a given rater pair can appear with either rater in the Scorer.x slot
  # depending on which column they happened to occupy for a given CDA.
  # Folding in the mirrored copy and then keeping only Scorer.x < Scorer.y
  # (by value) consolidates both into one consistent, value-ordered
  # direction per pair - shown once, not twice.
  sym_copy <- pairwise
  names(sym_copy)[names(sym_copy) %in% c("Scorer.x", "Scorer.y")] <- c("Scorer.y", "Scorer.x")
  names(sym_copy)[names(sym_copy) %in% c("Score.x", "Score.y")] <- c("Score.y", "Score.x")
  rbind(pairwise, sym_copy) %>%
    filter(as.integer(Scorer.x) < as.integer(Scorer.y))
}

# facet_grid always draws the full cross product of its two faceting
# variables' declared factor levels, regardless of whether a given
# combination has any data - so even with only the 45 upper-triangle rater
# pairs above, it would still draw all 100 panels (self-pairs + the
# now-empty lower-triangle mirror). This blanks those 55 non-pairs out
# entirely (rather than leaving them as visible empty bordered boxes), by
# replacing their panel grobs with nullGrob() after the plot is built.
hide_non_pairs <- function(p) {
  panel_info <- ggplot_build(p)$layout$layout
  hide_panels <- panel_info$PANEL[
    as.integer(as.character(panel_info$Scorer.x)) >= as.integer(as.character(panel_info$Scorer.y))
  ]

  g <- ggplotGrob(p)
  panel_idx <- grep("^panel", g$layout$name)
  stopifnot(length(panel_idx) == nrow(panel_info))
  for (pnum in as.integer(as.character(hide_panels))) {
    g$grobs[[panel_idx[pnum]]] <- grid::nullGrob()
  }
  g
}

make_patterns_plot <- function(sdl) {
  levels_xy <- as.character(1:10)
  score_levels <- as.character(0:6)

  # Self-pairs (a rater against themselves) carry no data and are excluded
  # outright by gen_pairwise_counts's Scorer.x < Scorer.y filter above.
  counts <- gen_pairwise_counts(sdl) %>%
    mutate(
      Scorer.x = factor(Scorer.x, levels = levels_xy),
      Scorer.y = factor(Scorer.y, levels = levels_xy),
      Score.x = factor(Score.x, levels = score_levels),
      Score.y = factor(Score.y, levels = score_levels)
    )

  # Complete the severity-score grid (0-6 x 0-6) *within* each rater pair
  # that actually shares data (nesting), rather than crossing all 10x10
  # rater combinations. The latter used to fabricate entirely-empty panels
  # (self-pairs, or raters who never scored together) whose Score.x/Score.y
  # were then all NA, which made ggplot's per-panel range() warn "no
  # non-missing arguments to min/max" for every such panel.
  complete_counts <- counts %>%
    complete(
      nesting(Scorer.x, Scorer.y),
      Score.x = factor(score_levels, levels = score_levels),
      Score.y = factor(score_levels, levels = score_levels),
      fill = list(Count = 0)
    ) %>%
    mutate(
      Count_Binned = cut(Count,
        breaks = c(0, 1, 10, 100, 250, Inf),
        labels = c("1", "2-10", "11-100", "101-250", "251-1000"),
        include.lowest = TRUE
      ),
      Scorer.y = fct_rev(Scorer.y)
    )

  p <- ggplot(complete_counts, aes(x = Score.x, y = Score.y)) +
    geom_tile(aes(fill = Count_Binned), color = "black", linewidth = 0.5) +
    # Score.x/Score.y are discrete factors (see above), so the exact-agreement
    # diagonal is drawn as a segment between the first and last factor
    # positions rather than geom_abline (which needs a continuous scale).
    geom_segment(

7. Krippendorff’s Alpha — Raw Scores

Code 
# Shared plotting helper for 7.1-7.3.
# The mean and threshold lines are geom_segments bounded to the bar heights
# (not full-height geom_vlines), so they physically cannot reach the label
# row above - the dashed/dotted line can't cover the text because it stops
# below it. Both labels sit at the same height, left-justified from their
# own line's x-position, growing rightward. The one exception: when the
# threshold (0.8) sits close enough to the mean that its left-justified
# label would run into the mean's label - as happens in 7.2's non-zero panel,
# where they differ by ~0.005 - the threshold label is right-justified
# instead, so it grows leftward, away from the mean label.
make_alpha_hist_panel <- function(boot_vec, xlab) {
  mean_val <- mean(boot_vec)
  threshold <- 0.8

  base <- ggplot(data.frame(alpha = boot_vec), aes(x = alpha)) +
    geom_histogram(binwidth = 0.005, fill = "cornflowerblue", color = "white")
  max_count <- max(ggplot_build(base)$data[[1]]$count)
  line_top <- max_count * 1.06
  label_y <- max_count * 1.15

  threshold_hjust <- if (abs(mean_val - threshold) < 0.02) 1 else 0

  base +
    geom_segment(
      x = mean_val, xend = mean_val, y = 0, yend = line_top,
      color = "firebrick", linetype = "dashed", linewidth = 1
    ) +
    geom_segment(
      x = threshold, xend = threshold, y = 0, yend = line_top,
      color = "black", linetype = "dotted", linewidth = 0.8
    ) +
    annotate("text",
      x = mean_val, y = label_y, hjust = 0, size = 4, color = "firebrick",
      label = sprintf("Mean=%.3f", mean_val)
    ) +
    annotate("text",
      x = threshold, y = label_y, hjust = threshold_hjust, size = 4, color = "black",
      label = "α=0.8"
    ) +
    scale_y_continuous(limits = c(0, max_count * 1.3), expand = expansion(mult = c(0.02, 0))) +
    coord_cartesian(clip = "off") +
    labs(x = xlab, y = "Count") +
    theme_minimal() +
    theme(
      axis.text = element_text(size = 13),
      axis.title = element_text(size = 15)
    )
}

# 7.1.1
# Computed on the full rater x CDA matrix directly (no median involved).
# Two distance metrics are compared here and in 7.3 as a robustness check on
# the assumed measurement level: ordinal (treats 0-6 as ordered categories)
# and interval (treats 0-6 as equally spaced). If the two estimates agree,
# the reliability conclusion doesn't hinge on which scale assumption is
# correct. Missing values are handled automatically: expected agreement is
# estimated from all rated pairs.
