Find median score and IoU-medoid representative box

Date: 21/07/2026

GOHREP

Goal:

Partner to median_and_centre_coords.qmd. Same cleaning and median-score steps, but the single representative box per CDA is chosen by IoU-medoid (the box with the highest mean intersection-over-union to the others) instead of the centre-of-overlap. The centre-of-overlap rule tends to pick the smallest box (a small box sits more centrally on the overlap), which occasionally selects a tiny/misplaced outlier; the medoid is the most-agreed-with box and is robust to one outlier. Cropped images are written to data/cropped_images_medoid/.

SCORE_COLS = ["Score1", "Score2", "Score3"]

# Step a) clean_coordinates nulls an individual rater's coordinates + score if the
# bounding box is degenerate (zero width/height) or has a missing coordinate. This
# removes annotations, not whole CDA rows.
cleaned_coords = data_combined.apply(dataset_preparation.clean_coordinates, axis=1)
n_coords_nulled = (
    data_combined[SCORE_COLS].notna().sum().sum()
    - cleaned_coords[SCORE_COLS].notna().sum().sum()
)

# Step b) clean_scores drops a whole CDA row if:
#   - >=2 of the 3 scores are missing, or
#   - exactly 1 score is missing and the remaining 2 disagree (no median possible).
n_missing = cleaned_coords[SCORE_COLS].isnull().sum(axis=1)
n_dropped_insufficient = int((n_missing >= 2).sum())
n_dropped_disagree = int(
    (
        (n_missing == 1)
        & cleaned_coords[SCORE_COLS].apply(lambda r: len(set(r.dropna())) > 1, axis=1)
    ).sum()
)

cleaned_data = dataset_preparation.clean_scores(cleaned_coords)

print(f"Original Data Length: {len(data_combined)}")
print(f"  Step a - annotations nulled (degenerate/missing bounding box): {n_coords_nulled}")
print(f"  Step b - CDAs dropped (fewer than 2 scores):                   {n_dropped_insufficient}")
print(f"  Step b - CDAs dropped (2 scores but they disagree):            {n_dropped_disagree}")
print(f"Cleaned Data Length: {len(cleaned_data)}")

6. Save cropped images (medoid box)

# 6
# Source images are read from data/raw_images_rotated/ (the shared-drive path used
# by save_cropped_images() in 05_save_cropped_images is offline; the rotated
# repo copies are coordinate-identical to the originals). Crop = raw[y1:y2, x1:x2]
# of the medoid box, saved to data/cropped_images_medoid/<Median_Score>/ with the
# same <Basename>_<Row>_<Col>_<Pos>.tif naming as the centre-of-overlap crops.
out_root = repo_root / "data" / "cropped_images_medoid"
raw_dir = repo_root / "data" / "raw_images_rotated"

to_crop = cleaned_data.dropna(subset=["Medoid_Coords", "Median_Score"]).copy()
to_crop["Medoid_Coords"] = to_crop["Medoid_Coords"].astype(int)

n_saved = 0
n_missing_src = 0
for basename, grp in to_crop.groupby("Basename"):
    src = raw_dir / basename
    raw_img = cv2.imread(str(src)) if src.exists() else None
    if raw_img is None:
        n_missing_src += len(grp)
        continue
    for _, row in grp.iterrows():
        m = int(row["Medoid_Coords"])
        x1, x2 = int(row[f"X1_{m}"]), int(row[f"X2_{m}"])
        y1, y2 = int(row[f"Y1_{m}"]), int(row[f"Y2_{m}"])
        crop = raw_img[y1:y2, x1:x2]
        score_dir = out_root / str(int(row["Median_Score"]))
        score_dir.mkdir(parents=True, exist_ok=True)
        fname = f"{os.path.splitext(row['Basename'])[0]}_{row['Row']}_{row['Col']}_{row['Pos']}.tif"
        cv2.imwrite(str(score_dir / fname), crop)
        n_saved += 1

print(f"Saved {n_saved} cropped images to {out_root}")
print(f"CDAs skipped (source image unavailable): {n_missing_src}")
