PCA of CDA image dataset

Date: 18/3/24 and 25/3/24

GOHREP

Goal:

  1. See how images cluster together following PCA Dimensionality Reduction
  2. Identify images that cluster most distantly from others of their score

Hypothesis:

  1. Images of the same score will cluster together
  2. Images of adjacent scores will cluster closer than those of more distant scores
  3. Some images will cluster far from other images of the same score, which may indicate mislabelling

Rationale:

Images of the same score are likely to be more similar than images of different scores, and images of adjacent scores are likely to be more similar than those of distant scores.

Due to variation in human scoring, even with the median of three scores taken, there may be human errors or biases which result in image misclassifications. However, these images that appear to be misclassified may instead just be uncommon.

# Undersampling rationale: PCA does not use any class labels to reduce the
# dimensionality, but it acts on the dataset as a whole to maximise the
# variance explained by any eigenvectors. If the classes are unbalanced, the
# PCA may be biased towards explaining variance in the larger classes (such
# as 0). We want the PCA to capture the underlying variance of all classes
# more equally, which gives a rationale for undersampling. The same is true
# in the k-nearest neighbour clustering: if one group is much larger (e.g.
# 0s), in the PCA regions where that group overlaps with others (e.g. 1s or
# 2s), the closest neighbouring points to any given point in that region
# will likely be from the larger group, biasing the predictions towards that
# larger group.
data_dir = str(repo_root / "data" / "cropped_images")
image_size = 64
def load_images_and_labels() -> Tuple[List[Any], List[int], List[Any]]:
    # List the possible labels (0-6)
    class_labels = os.listdir(data_dir)
    class_labels = [label for label in class_labels if os.path.isdir(os.path.join(data_dir, label))]

    images = []
    labels = []
    filepaths = []

    for class_label in class_labels:
        class_dir = os.path.join(data_dir, class_label)
        image_files = os.listdir(class_dir) # List of CDA images in current score folder

        for image_file in image_files:
            image_path = os.path.join(class_dir, image_file)
            if os.path.isfile(image_path) and image_path.endswith(".DS_Store"): # Ignore .DS_Store files
                continue
            image = cv.imread(image_path)
            image = cv.resize(image, (image_size, image_size)) # Preprocessing the cropped images to be the same size.

            images.append(image)
            labels.append(class_label)
            filepaths.append(image_path)

    return images, labels, filepaths

def downsample_oversized_classes(train_images, train_labels, train_filepaths, per_class_train_size):
    # Undersample the classes larger than per_class_train_size
    undersampled_images, undersampled_labels, undersampled_filepaths = [], [], []
    for label in np.unique(train_labels):
        samples_indices = np.where(train_labels == label)[0]
        if len(samples_indices) > per_class_train_size:
            samples_indices = resample(samples_indices, replace=False, n_samples=per_class_train_size, random_state=42)
        undersampled_images.extend(train_images[samples_indices])
        undersampled_labels.extend(train_labels[samples_indices])
        undersampled_filepaths.extend(train_filepaths[samples_indices])

    # Shuffle the data
    undersampled_images = np.array(undersampled_images)
    undersampled_labels = np.array(undersampled_labels)
    undersampled_filepaths = np.array(undersampled_filepaths)
    shuffle_indices = np.random.permutation(len(undersampled_labels))
    undersampled_images = undersampled_images[shuffle_indices]
    undersampled_labels = undersampled_labels[shuffle_indices]
    undersampled_filepaths = undersampled_filepaths[shuffle_indices]

    return undersampled_images, undersampled_labels, undersampled_filepaths

def implement_pca(images, n_components=10):
    images_array = np.array(images)
    images_array = images_array.reshape(len(images_array), -1)

    pca = PCA(n_components=n_components)
    pca.fit(images_array)
    images_pca = pca.transform(images_array)

    return pca, images_pca

def plot_eigenvalues(pca, name):

    # the difference is class balance, not which scores are included:
    # "original" keeps the natural (0-heavy) class distribution, while

    plt.ylabel('Variance Explained')
    plt.title(title)

5.1 kNN implementation

# 5.1
from sklearn.metrics import silhouette_score, adjusted_rand_score, normalized_mutual_info_score, fowlkes_mallows_score, confusion_matrix
from sklearn.neighbors import NearestNeighbors
from matplotlib.patches import Patch


def nearest_neighbour_clustering(true_labels, images_pca, name, select_by='silhouette'):
    # select_by picks which criterion chooses best_k out of k=1..10:
    #  - 'silhouette': unsupervised - how well-separated the resulting
    #    majority-vote groups are in PCA space; never looks at true_labels.
    #  - 'accuracy': supervised - the leave-one-out exact accuracy of the
    #    majority-vote prediction against true_labels directly.
    best_k = None
    best_silhouette_score = -1  # Initialize with a low value
    best_ari = -1  # Initialize with a low value
    best_loo_exact_acc = -1  # Initialize with a low value

    for k in range(1, 11):
        # Initialize the k-nearest neighbors model
        nbrs = NearestNeighbors(n_neighbors=k, metric='euclidean')

        # Fit the model to your PCA-transformed images
        nbrs.fit(images_pca)

        # kneighbors() with no argument excludes each point from its own neighbourhood,
        # giving a leave-one-out evaluation equivalent to a held-out test for every image.
        distances, indices = nbrs.kneighbors()

        # Calculate labels based on nearest neighbors
        knn_labels = np.zeros_like(true_labels)
        for i, neighbors in enumerate(indices):
            neighbor_labels = true_labels[neighbors]
            knn_labels[i] = np.bincount(neighbor_labels).argmax()

        # Compute silhouette score
        silhouette = silhouette_score(images_pca, knn_labels)

        # Calculate ARI
        ari = adjusted_rand_score(true_labels, knn_labels)

        # Leave-one-out exact accuracy at this k, used only when select_by='accuracy'.
        loo_exact_acc = np.mean(knn_labels == true_labels)

        criterion = silhouette if select_by == 'silhouette' else loo_exact_acc
        best_criterion_so_far = best_silhouette_score if select_by == 'silhouette' else best_loo_exact_acc
        if criterion > best_criterion_so_far:
            best_k = k
            best_silhouette_score = silhouette
            best_ari = ari
            best_loo_exact_acc = loo_exact_acc

    print(f"Selecting k by: {select_by}")
    print(f"Best k value: {best_k}")
    print(f"Best silhouette score: {best_silhouette_score}")
    print(f"Adjusted Rand Index (ARI) for best k: {best_ari}")

    # Refit at best_k specifically. The search loop above overwrites
    # distances/indices every iteration, so without this refit they would
    # silently hold k=10's (the last value the loop tried) neighbours below,
    # regardless of which k was actually selected as best - a bug in the
    # original version of this function, where every downstream result
    # (predicted_labels, accuracy, confusion matrix, distance heatmap) used
    # k=10 even when best_k was reported as something else.
    nbrs = NearestNeighbors(n_neighbors=best_k, metric='euclidean')
    nbrs.fit(images_pca)
    distances, indices = nbrs.kneighbors()

    # Predicted labels using majority voting among the k nearest neighbors
    predicted_labels = np.zeros_like(true_labels)

    for i in range(len(images_pca)):
        neighbor_labels = true_labels[indices[i]]
        predicted_labels[i] = np.bincount(neighbor_labels).argmax()

    exact_acc = np.mean(predicted_labels == true_labels)
    print(f"Exact accuracy: {exact_acc:.3f}")

    near_miss_acc = np.mean(np.abs(predicted_labels - true_labels) <= 1)
    print(f"Near-miss accuracy (within +/-1): {near_miss_acc:.3f}")

    unique_labels = np.unique(true_labels)
    per_class_exact_acc = {label: np.mean(predicted_labels[true_labels == label] == label)
                     for label in unique_labels}
    print(f"Per-class exact accuracy: { {k: round(v, 3) for k, v in per_class_exact_acc.items()} }")

    per_class_near_miss_acc = {label: np.mean(np.abs(predicted_labels[true_labels == label] - label) <= 1)
                     for label in unique_labels}
    print(f"Per-class near-miss accuracy: { {k: round(v, 3) for k, v in per_class_near_miss_acc.items()} }")

    unique_labels = np.unique(true_labels)

    mean_distances_per_label = []
    unique_labels = np.unique(true_labels)

    for label in unique_labels:
        mask = true_labels == label
        distances_for_label = distances[mask]  # Subset of distances for the current label
        mean_distance_for_label = np.mean(distances_for_label)  # Calculate mean distance for the current label
        mean_distances_per_label.append(mean_distance_for_label)

    print(f"Mean distance to neighbours for each score {mean_distances_per_label}")

    class_data = {}  # Dictionary to store data for each class
    for label in unique_labels:
        mask = true_labels == label
        class_data[label] = images_pca[mask]  # Subset of data for the current label

    # Calculate distances between each class
    class_distances = {}
    for label1, data1 in class_data.items():
        for label2, data2 in class_data.items():
            if label1 != label2:
                # Compute distances between each pair of classes
                nbrs.fit(data2)
                distances, _ = nbrs.kneighbors(data1, n_neighbors=1)
                mean_distance = np.mean(distances)
                class_distances[(label1, label2)] = mean_distance

    # Within-class distances, on the diagonal: same methodology as the
    # between-class distances above (mean nearest-neighbour distance per
    # point), just fit and queried on a class's own data. kneighbors() with
    # no argument excludes each point from being its own neighbour (same
    # trick used earlier in this function), so this is each point's distance
    # to the nearest OTHER point of its own class, not to itself.
    class_within_distances = {}
    for label, data in class_data.items():
        if len(data) < 2:
            continue
        within_nbrs = NearestNeighbors(n_neighbors=1, metric='euclidean')
        within_nbrs.fit(data)
        distances, _ = within_nbrs.kneighbors()
        class_within_distances[label] = np.mean(distances)

    def plot_class_distances(class_distances, class_within_distances):
        unique_labels = np.unique([label for labels in class_distances.keys() for label in labels])
        n = len(unique_labels)

        # Between- and within-class distances share one colour scale (same
        # vmin/vmax across both matrices) so equal distances read as equal
        # colour intensity, despite being drawn with two different
        # colourmaps (green = between, blue = within) on the same axes.
        between_matrix = np.full((n, n), np.nan)
        within_matrix = np.full((n, n), np.nan)
        for (label1, label2), distance in class_distances.items():
            between_matrix[label1, label2] = distance
        for label, distance in class_within_distances.items():
            within_matrix[label, label] = distance

        vmin = np.nanmin([np.nanmin(between_matrix), np.nanmin(within_matrix)])
        vmax = np.nanmax([np.nanmax(between_matrix), np.nanmax(within_matrix)])

        plt.figure(figsize=(9, 6))
        im_between = plt.imshow(between_matrix, cmap='Greens', vmin=vmin, vmax=vmax, interpolation='nearest')
        im_within = plt.imshow(within_matrix, cmap='Blues', vmin=vmin, vmax=vmax, interpolation='nearest')

        # Add annotations
        for i in range(n):
            for j in range(n):
                value = within_matrix[i, j] if i == j else between_matrix[i, j]
                if not np.isnan(value):
                    plt.text(j, i, f"{value:.2f}", ha='center', va='center', color='black')

        plt.xticks(np.arange(n), unique_labels)
        plt.yticks(np.arange(n), unique_labels)
        plt.xlabel('Class')
        plt.ylabel('Class')
        plt.title('Mean Nearest-Neighbour Distance: Between- vs Within-Class')

        # Two colourbars, slotted side by side, sharing one vmin/vmax (set
        # above). Only the labels are hidden on the within-class one (not
        # its ticks - calling set_ticks() with the between-class bar's tick
        # values breaks the Blues gradient into discrete blocks) since both
        # bars already show the same scale.
        cbar_between = plt.colorbar(im_between, label='Mean distance to nearest neighbour', fraction=0.046, pad=0.04)
        cbar_within = plt.colorbar(im_within, fraction=0.046, pad=0.08)
        cbar_within.ax.set_yticklabels([])

        plt.legend(
            handles=[
                Patch(facecolor=plt.cm.Greens(0.7), edgecolor='black', label='Between-class'),
                Patch(facecolor=plt.cm.Blues(0.7), edgecolor='black', label='Within-class'),
            ],
            loc='upper left', bbox_to_anchor=(1.6, 0.3)
        )
        plt.savefig(str(plots_dir / f"knn_class_distances_{name}.png"), dpi=300, bbox_inches='tight')
        plt.show()

    plot_class_distances(class_distances, class_within_distances)

    # Scatter plot of predicted cluster assignments
    plt.figure(figsize=(8, 6))
    unique_labels_pred = np.unique(predicted_labels)
    colors_pred = plt.cm.tab10(np.linspace(0, 1, len(unique_labels_pred)))
    for i, label in enumerate(unique_labels_pred):
        mask = predicted_labels == label
        plt.scatter(images_pca[mask, 0], images_pca[mask, 1], color=colors_pred[i], label=f'{label}', s=10, alpha=0.5)
    plt.xlabel('PC1')
    plt.ylabel('PC2')
    plt.legend(title='Severity Score', loc='upper left', bbox_to_anchor=(1, 1))
    plt.savefig(str(plots_dir / f"knn_predictions_{name}.png"), dpi=300, bbox_inches='tight')

7. Additional Information

7.1 How does PCA work?

PCA is a dimensionality reduction technique, aiming to capture the most important information in the data and minimizing redundancy and noise.

  1. Build a covariance matrix which captures the relationships between pairs of variables (in our case variables are pixel values). A high covariance suggests a strong relationship.
  2. Calculate eigenvectors, which are directions in the original multidimensional feature space, representing a PC. Calculate eigenvalues, which are the magnitude of variance captured by the corresponding eigenvector.
  3. Rank PCs by their eigenvalues.
  4. Transform the input data into a new coordinate system defined by the principal components.

For each image, the total number of variables is: 64 x 64 x 3 = 12,288 Given this large number of variables, there is likely to be more variability in the data and capturing this with a small number of principal components is more challenging.
