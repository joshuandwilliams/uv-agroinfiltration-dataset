results_path = repo_root / "analyses" / "03_human_performance" / "human_performance_results.csv"


def scorer_accuracy_table(scorer_cms, labels):
    """Per-rater Exact/Near-Miss Accuracy (%) from a dict of {rater: confusion_matrix}."""
    rows = []
    for scorer, cm in scorer_cms.items():
        acc = human_performance.cm_accuracies(cm, labels)
        rows.append({
            "Rater": scorer,
            "Exact Accuracy": acc["exact_accuracy"] * 100,
            "Near-Miss Accuracy": acc["within_one_accuracy"] * 100,
        })
    return pd.DataFrame(rows)


def pairwise_scorer_accuracy_table(data_wide):
    """Per-rater Exact/Near-Miss Accuracy (%) for Metric 2, from the wide-format score table."""
    pairs = []
    for i, j in [(1, 2), (1, 3), (2, 3)]:
        sub = data_wide[[f"Scorer{i}", f"Score{i}", f"Scorer{j}", f"Score{j}"]].copy()
        sub.columns = ["ScorerA", "ScoreA", "ScorerB", "ScoreB"]
        pairs.append(sub)
    pairs_df = pd.concat(pairs).dropna(subset=["ScoreA", "ScoreB"])
    pairs_df["Diff"] = (pairs_df["ScoreA"] - pairs_df["ScoreB"]).abs()

    sym = pd.concat([
        pairs_df[["ScorerA", "Diff"]].rename(columns={"ScorerA": "Rater"}),
        pairs_df[["ScorerB", "Diff"]].rename(columns={"ScorerB": "Rater"}),
    ])
    return (
        sym.groupby("Rater")["Diff"]
        .agg(**{
            "Exact Accuracy": lambda d: (d == 0).mean() * 100,
            "Near-Miss Accuracy": lambda d: (d <= 1).mean() * 100,
        })
        .reset_index()
    )

        data_long = pd.wide_to_long(data_normclass, stubnames=['Scorer', 'Score'], i=['Basename', 'Row', 'Col', 'Pos'], j='num').reset_index()
        data_long_noNA = data_long.dropna(subset=['Score'])

        # Metric 1: Agreement with the Median Severity Score

def get_dist_stats(dist, name):
    mean_agreement = np.mean(dist)
    std_agreement = np.std(dist)
    sem_agreement = st.sem(dist)
    ci_agreement = st.t.interval(confidence=0.95, df=len(dist)-1, loc=mean_agreement, scale=sem_agreement)
    ci_lower, ci_upper = ci_agreement
    ci_diff = (ci_upper - ci_lower) / 2
    print(name)
    print(f"Mean: {mean_agreement:.5f}")
    print(f"95% CI: {ci_diff:.5f}, ({ci_agreement[0]:.5f}, {ci_agreement[1]:.5f})")
    print(f"SD: {std_agreement:.5f}\n")

def plot_distribution(dist, name, x_label):
    mean_agreement = np.mean(dist)

    plt.figure(figsize=(10, 6))
    plt.hist(dist, bins='auto', alpha=0.75, color='skyblue', edgecolor='black')
    plt.axvline(mean_agreement, color='red', linestyle='--', linewidth=2, label=f'Mean: {mean_agreement:.5f}')
    plt.xlabel(x_label)
    plt.ylabel('Frequency')
    plt.legend(loc='upper right')
    plt.grid(axis='y', linestyle='--', alpha=0.7)
    plt.savefig(name)
    plt.show()
