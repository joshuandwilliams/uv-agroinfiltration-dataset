2. Compute agreement metrics

Metric 1: Rater agreement with their own median - Treat the model and rater the same: compare both against the rater median. - Rater scores are compared to a median they helped build, so human performance is inflated. - All agree: 100% - 2/3 agree: 67% - None agree: 33% <- problem

Metric 2: Average agreement of each score pair - A measure of inter-rater agreement. - All agree: 100% - 2/3 agree: 33% <- problem - None agree: 0%

Metric 3: Corrected rater agreement with their own median - Adapt metric 1 by setting agreement = 0 for three disagreeing scores. This removes the “guaranteed agreement” issue of Metric 1. - All agree: 100% - 2/3 agree: 67% - None agree: 0%

Metric 3 appears to be the most appropriate metric.

cache_dir = repo_root / "analyses" / "03_human_performance"
metric1_path = cache_dir / "metric1_resampling.csv"

def scorer_accuracy_table(scorer_cms, labels):
    """Per-rater Exact/Near-Miss Accuracy (%) from a dict of {rater: confusion_matrix}."""
    rows = []
    for scorer, cm in scorer_cms.items():
        acc = human_performance.cm_accuracies(cm, labels)
        rows.append({
            "Rater": scorer,
            "Exact Accuracy": acc["exact_accuracy"] * 100,
            "Near-Miss Accuracy": acc["within_one_accuracy"] * 100,
        })
    return pd.DataFrame(rows)


def pairwise_scorer_accuracy_table(data_wide):
    """Per-rater Exact/Near-Miss Accuracy (%) for Metric 2, from the wide-format score table."""
    pairs = []
    for i, j in [(1, 2), (1, 3), (2, 3)]:
        sub = data_wide[[f"Scorer{i}", f"Score{i}", f"Scorer{j}", f"Score{j}"]].copy()
        sub.columns = ["ScorerA", "ScoreA", "ScorerB", "ScoreB"]
        pairs.append(sub)
    pairs_df = pd.concat(pairs).dropna(subset=["ScoreA", "ScoreB"])
    pairs_df["Diff"] = (pairs_df["ScoreA"] - pairs_df["ScoreB"]).abs()

    sym = pd.concat([
        pairs_df[["ScorerA", "Diff"]].rename(columns={"ScorerA": "Rater"}),
        pairs_df[["ScorerB", "Diff"]].rename(columns={"ScorerB": "Rater"}),
    ])
    return (
        sym.groupby("Rater")["Diff"]
        .agg(**{
            "Exact Accuracy": lambda d: (d == 0).mean() * 100,
            "Near-Miss Accuracy": lambda d: (d <= 1).mean() * 100,
        })
        .reset_index()
    )


def corrected_scorer_accuracy_table(corrected_long):
    """Per-rater Exact/Near-Miss Accuracy (%) for Metric 3, from the corrected long-format table."""
    return (
        corrected_long.groupby("Scorer")
        .agg(**{
            "Exact Accuracy": ("Exact_Match", lambda s: s.mean() * 100),
            "Near-Miss Accuracy": ("NearMiss_Match", lambda s: s.mean() * 100),
        })
        .reset_index()
        .rename(columns={"Scorer": "Rater"})
    )


def pattern_breakdown_table(corrected_long):
    """Proportion of CDAs per rater-agreement pattern, and raw (Metric 1) vs
    corrected (Metric 3) exact-agreement-with-median rate within each pattern -
    shows exactly where the correction changes the raw figures."""
    cda_id = ["Basename", "Row", "Col", "Pos"]
    pattern_per_cda = corrected_long.groupby(cda_id)["Agreement_Pattern"].first()
    pattern_share = (pattern_per_cda.value_counts(normalize=True) * 100).rename("% of CDAs")

    rates = corrected_long.groupby("Agreement_Pattern").agg(**{
        "Raw Metric 1 (%)": ("Raw_Exact_Match", lambda s: s.mean() * 100),
        "Corrected Metric 3 (%)": ("Exact_Match", lambda s: s.mean() * 100),
    })

    order = ["All agree", "2 agree, 1 different", "All different", "Fewer than 3 raters"]
    return (
        pd.concat([pattern_share, rates], axis=1)
        .reindex([p for p in order if p in pattern_share.index])
        .reset_index()
        .rename(columns={"index": "Agreement Pattern"})
    )


# Each metric has its own cache, so adding/changing one metric never forces
# a recompute of the others - only the missing ones re-run the 1000-iteration
# resampling loop below (and only their own block within it).
needs_metric1 = not metric1_path.exists()
needs_metric2 = not metric2_path.exists()
needs_metric3 = not metric3_path.exists()
needs_metric3_nz = not metric3_nonzero_path.exists()

if needs_metric1 or needs_metric2 or needs_metric3 or needs_metric3_nz:
    missing = [name for name, needed in [
        ("Metric 1", needs_metric1), ("Metric 2", needs_metric2), ("Metric 3", needs_metric3),
        ("Metric 3 (non-median-0)", needs_metric3_nz),
    ] if needed]
    print(f"Running 1000-iteration resampling loop for: {', '.join(missing)} (this will take a while)...")
    num_iterations = 1000
    agreement_dist, agreement_nearmiss = [], []
    pairwise_dist, pairwise_nearmiss = [], []
    corrected_dist, corrected_nearmiss = [], []
    corrected_nz_dist, corrected_nz_nearmiss = [], []

    for i in range(1, num_iterations + 1):
        if i % 100 == 0:
            print(f"Iteration (seed): {i} / {num_iterations}")

        # Downsample score classes
        data_normclass = human_performance.downsample_score_classes_csv(data, seed=i)

        if needs_metric1 or needs_metric3:
            data_long = pd.wide_to_long(data_normclass, stubnames=['Scorer', 'Score'], i=['Basename', 'Row', 'Col', 'Pos'], j='num').reset_index()
            data_long_noNA = data_long.dropna(subset=['Score'])

        if needs_metric1:
            # Metric 1: Agreement with the Median Severity Score
            full_cm = confusion_matrix(data_long_noNA['Median_Score'], data_long_noNA['Score'])
            accuracies = human_performance.cm_accuracies(full_cm, np.sort(data_long_noNA['Score'].unique()))
            agreement_dist.append(accuracies['exact_accuracy'])
            agreement_nearmiss.append(accuracies['within_one_accuracy'])

        if needs_metric2:
            # Metric 2: Agreement between score pairs (vectorised - avoids the
            # per-row .apply() cost of iterating find_pairwise_agreement)
            pairwise_dist.append(human_performance.pairwise_agreement_rate(data_normclass, precision=0).mean())
            pairwise_nearmiss.append(human_performance.pairwise_agreement_rate(data_normclass, precision=1).mean())

        if needs_metric3:
            # Metric 3: Metric 1, corrected for the median-of-3 tautology
            corrected_long_iter = human_performance.corrected_median_agreement(data_long_noNA)
            corrected_dist.append(corrected_long_iter["Exact_Match"].mean())
            corrected_nearmiss.append(corrected_long_iter["NearMiss_Match"].mean())

        if needs_metric3_nz:
            # Metric 3, restricted to CDAs whose median severity score is not 0.
            # Class-0 CDAs are near-unanimously agreed upon, so excluding them
            # isolates agreement on the harder, non-zero severity classes -
            # mirroring the non-median-0 comparison used for Krippendorff's alpha.
            data_nz = data[data["Median_Score"] != 0]
            data_nz_normclass = human_performance.downsample_score_classes_csv(data_nz, seed=i)
            data_nz_long = pd.wide_to_long(data_nz_normclass, stubnames=['Scorer', 'Score'], i=['Basename', 'Row', 'Col', 'Pos'], j='num').reset_index()
            data_nz_long_noNA = data_nz_long.dropna(subset=['Score'])
            corrected_nz_long_iter = human_performance.corrected_median_agreement(data_nz_long_noNA)
            corrected_nz_dist.append(corrected_nz_long_iter["Exact_Match"].mean())
            corrected_nz_nearmiss.append(corrected_nz_long_iter["NearMiss_Match"].mean())

    if needs_metric1:
        pd.DataFrame({"agreement_dist": agreement_dist, "agreement_nearmiss": agreement_nearmiss}).to_csv(metric1_path, index=False)
    if needs_metric2:
        pd.DataFrame({"pairwise_dist": pairwise_dist, "pairwise_nearmiss": pairwise_nearmiss}).to_csv(metric2_path, index=False)
    if needs_metric3:
        pd.DataFrame({"corrected_dist": corrected_dist, "corrected_nearmiss": corrected_nearmiss}).to_csv(metric3_path, index=False)
    if needs_metric3_nz:
        pd.DataFrame({"corrected_nz_dist": corrected_nz_dist, "corrected_nz_nearmiss": corrected_nz_nearmiss}).to_csv(metric3_nonzero_path, index=False)

if not needs_metric1:

if not needs_metric2:

def get_dist_stats(dist, name):
    mean_agreement = np.mean(dist)
    std_agreement = np.std(dist)
    sem_agreement = st.sem(dist)
    ci_agreement = st.t.interval(confidence=0.95, df=len(dist)-1, loc=mean_agreement, scale=sem_agreement)
    ci_lower, ci_upper = ci_agreement
    ci_diff = (ci_upper - ci_lower) / 2
    print(name)
    print(f"Mean: {mean_agreement:.5f}")
    print(f"95% CI: {ci_diff:.5f}, ({ci_agreement[0]:.5f}, {ci_agreement[1]:.5f})")
    print(f"SD: {std_agreement:.5f}\n")

4. Distribution plots

Repeat stratified undersampling: Each of the 1,000 iterations draws a random subsample of 436 CDAs for each severity class (undersampling each class down to the smallest one’s size). Undersampling means that no single class can dominate the metric. Repeating many times at random makes sure that the randomly chosen subset isn’t isn’t especially lucky or unlucky by chance.

# 4.1
def plot_distribution(dist, name, x_label):
    mean_agreement = np.mean(dist)

    plt.figure(figsize=(10, 6))
    plt.hist(dist, bins='auto', alpha=0.75, color='skyblue', edgecolor='black')
    plt.axvline(mean_agreement, color='red', linestyle='--', linewidth=2, label=f'Mean: {mean_agreement:.5f}')
    plt.xlabel(x_label)
    plt.ylabel('Frequency')
    plt.legend(loc='upper right')
    plt.grid(axis='y', linestyle='--', alpha=0.7)
    plt.savefig(name)
    plt.show()
